   ```typescript
   import { writeFileSync } from 'node:fs';
   import { join } from 'node:path';
   import { MemoryStore } from '../core/memory-store.js';

   export interface RetroInput {
     summary: string;
     learnings: string;
     metrics: Record<string, unknown>;
   }

   export function writeRetrospective(
     store: MemoryStore,
     sprintId: string, // 'sprint-NNN' format
     projectRoot: string,
     input: RetroInput,
   ): void {
     const sprintNum = sprintId.replace('sprint-', '');
     const now = new Date().toISOString();
     // DB upsert: 3 rows
     store.upsert({
       id: `sprint-log-${sprintNum}`,
       type: 'sprint',
       sprint_id: sprintId,
       title: `Sprint ${sprintNum} log`,
       body: JSON.stringify(input.metrics),
       status: 'completed',
       updated_at: now,
     });
     store.upsert({
       id: `retro-${sprintId}`,
       type: 'retro',
       sprint_id: sprintId,
       title: `Sprint ${sprintNum} retrospective`,
       body: input.summary,
       status: 'accepted',
       updated_at: now,
     });
     store.upsert({
       id: `mem-${sprintId}`,
       type: 'memory',
       sprint_id: sprintId,
       title: `Sprint ${sprintNum} learnings`,
       body: input.learnings,
       status: 'active',
       updated_at: now,
     });
     // FS: RETRO.md atomic write
     const retroMd = renderRetroMd(input, sprintId, now);
     writeFileSync(join(projectRoot, '.brain', 'RETRO.md'), retroMd);
   }

   function renderRetroMd(input: RetroInput, sprintId: string, timestamp: string): string {
     return `# Retrospective — ${sprintId}\n\n_Updated: ${timestamp}_\n\n## Summary\n${input.summary}\n\n## Learnings\n${input.learnings}\n\n## Metrics\n${JSON.stringify(input.metrics, null, 2)}\n`;
   }
   ```
