```typescript
// src/core/adr-file-sync.ts
import { readdir, readFile } from 'node:fs/promises';
import { join } from 'node:path';
import type Database from 'better-sqlite3';

const MADR_TITLE_REGEX = /^#\s+ADR-(\d+)\s+(.+)$/m;
const MADR_STATUS_REGEX = /^\*\*Status:\*\*\s+(\w+)/mi;
const MADR_DATE_REGEX = /^\*\*Date:\*\*\s+(\d{4}-\d{2}-\d{2})/mi;

export async function syncAdrFilesToDb(db: Database.Database, adrDir: string): Promise<number> {
  const files = (await readdir(adrDir)).filter(f => /^\d{3}-/.test(f) && f.endsWith('.md'));
  let count = 0;
  const upsert = db.prepare(`INSERT INTO entries (id, type, status, title, date, content)
    VALUES (?, 'adr', ?, ?, ?, ?)
    ON CONFLICT(id) DO UPDATE SET status=excluded.status, title=excluded.title, content=excluded.content`);
  for (const file of files) {
    const content = await readFile(join(adrDir, file), 'utf-8');
    const titleM = content.match(MADR_TITLE_REGEX);
    const statusM = content.match(MADR_STATUS_REGEX);
    const dateM = content.match(MADR_DATE_REGEX);
    if (!titleM) continue;
    const id = `adr-${titleM[1]}`;
    upsert.run(id, statusM?.[1] ?? 'proposed', titleM[2], dateM?.[1] ?? null, content);
    count++;
  }
  return count;
}
```

```typescript
// src/orchestra/managed-docs/doc-cache.ts
export function computeCacheKey(input: { fileHash: string; entryHash: string; sprintId?: string }): string {
  const { fileHash, entryHash, sprintId } = input;
  if (!sprintId) return `${fileHash}:${entryHash}`;  // BACKWARDS COMPAT
  return `${fileHash}:${entryHash}:${sprintId}`;
}
```
