// src/orchestra/sprint-pid-manager.ts
import { writeFileSync, readFileSync, existsSync, renameSync, unlinkSync, mkdirSync } from 'node:fs';
import { join, dirname } from 'node:path';

export interface SprintStateSnapshot {
  sprintId: string;
  pid: number;
  startedAt: string;
  currentWave: number;
  taskStatuses: Record<string, string>;
  metricsJsonlSize: number;
  lastHeartbeat: string;
}

export interface OrphanInfo {
  sprintId: string;
  pid: number;
  stateSnapshot: SprintStateSnapshot | null;
  pidFile: string;
  stateFile: string;
}

function pidPath(root: string, sprintId: string): string {
  return join(root, '.deckent', `${sprintId}.pid`);
}

function statePath(root: string, sprintId: string): string {
  return join(root, '.deckent', `${sprintId}.state.json`);
}

export function writePid(root: string, sprintId: string): void {
  const p = pidPath(root, sprintId);
  mkdirSync(dirname(p), { recursive: true });
  writeFileSync(p, String(process.pid));
}

export function readPid(root: string, sprintId: string): number | null {
  const p = pidPath(root, sprintId);
  if (!existsSync(p)) return null;
  const n = Number(readFileSync(p, 'utf-8').trim());
  return Number.isFinite(n) ? n : null;
}

export function clearPid(root: string, sprintId: string): void {
  const p = pidPath(root, sprintId);
  if (existsSync(p)) unlinkSync(p);
}
```

export function writeStateSnapshot(root: string, sprintId: string, snap: SprintStateSnapshot): void {
  const final = statePath(root, sprintId);

  } catch {
    return null; // corrupt snapshot → caller handles
  }

function isProcessAlive(pid: number): boolean {
  try {
    process.kill(pid, 0);
    return true;
  } catch (e: any) {
    return e.code === 'EPERM'; // EPERM means exists but no permission → alive
  }
}

export function detectOrphan(root: string, sprintId: string): OrphanInfo | null {
  const pid = readPid(root, sprintId);
  if (pid === null) return null;
  if (isProcessAlive(pid)) return null;
  return {
    sprintId,
    pid,
    stateSnapshot: readStateSnapshot(root, sprintId),
    pidFile: pidPath(root, sprintId),
    stateFile: statePath(root, sprintId),
  };
}

import { existsSync, readFileSync } from 'node:fs';
import { join } from 'node:path';

const DONE_SET = new Set(['DONE', 'GO_WITH_TECH_DEBT']);

export function shouldReportStale(
  projectRoot: string,
  taskId: string,
  hb: { status?: string; exitCode?: number } | null,
): boolean {
  const resultPath = join(projectRoot, '.tasks', `task-${taskId}.result`);
  if (!existsSync(resultPath)) return true; // no result → normal stale detection
  try {
    const result = JSON.parse(readFileSync(resultPath, 'utf-8'));
    if (DONE_SET.has(result.selfAssessment)) return false;
    return true; // NO_GO or malformed → honest stale
  } catch {
    return true; // parse fail → honest stale (fail-safe)
  }
}
```

export function finalizeHeartbeatOnShutdown(projectRoot: string, taskId: string): void {
  const resultPath = join(projectRoot, '.tasks', `task-${taskId}.result`);
  const hbPath = join(projectRoot, '.tasks', `task-${taskId}.hb`);
  if (!existsSync(resultPath)) return; // no result → honest FAILED stays
  try {
    const result = JSON.parse(readFileSync(resultPath, 'utf-8'));
    if (result.selfAssessment === 'DONE' || result.selfAssessment === 'GO_WITH_TECH_DEBT') {
      const hb = existsSync(hbPath) ? JSON.parse(readFileSync(hbPath, 'utf-8')) : {};
      hb.status = 'DONE';
      hb.exitCode = 0;
      hb.timestamp = new Date().toISOString();
      writeFileSync(hbPath, JSON.stringify(hb));
    }
  } catch {
    // fail-safe: leave HB alone
  }
}

// Register SIGTERM handler in worker main entry:
process.on('SIGTERM', () => {
  const taskId = process.env.DECKENT_TASK_ID;
  const root = process.env.DECKENT_PROJECT_ROOT;
  if (taskId && root) finalizeHeartbeatOnShutdown(root, taskId);
  process.exit(0);
});
```

import { exec } from 'node:child_process';
import { promisify } from 'node:util';
const execAsync = promisify(exec);

export interface VerifyResult { ok: boolean; reason?: string; attempts: number; }

export async function enforceVerifyLoop(projectRoot: string, taskId: string, scope: string): Promise<VerifyResult> {
  const MAX_ATTEMPTS = 3;
  const TIMEOUT_MS = 300_000;
  for (let attempt = 1; attempt <= MAX_ATTEMPTS; attempt++) {
    try {
      await execAsync('npx tsc --noEmit', { cwd: projectRoot, timeout: TIMEOUT_MS });
      await execAsync(`npx vitest run ${scope}`, { cwd: projectRoot, timeout: TIMEOUT_MS });
      mkdirSync(join(projectRoot, '.tasks'), { recursive: true });
      writeFileSync(join(projectRoot, '.tasks', `${taskId}.verify-ran`), new Date().toISOString());
      return { ok: true, attempts: attempt };
    } catch (e: any) {
      if (attempt === MAX_ATTEMPTS) {
        return { ok: false, reason: e.message || 'verify_loop_exhausted', attempts: attempt };
      }
    }
  }
  return { ok: false, reason: 'verify_loop_exhausted', attempts: MAX_ATTEMPTS };
}
```

```typescript
const DECAY_EXEMPT = new Set<string>(['DECISIONS.md', 'PROJECT-IDENTITY.md']);

export interface BrainBudgetAudit {
  decayableLines: number;
  permanentLines: number;
  totalLines: number;
  budget: number;
  status: 'OK' | 'OVER';
  recommendation: string | null;
}

export function auditBrainBudget(projectRoot: string, budget: number): BrainBudgetAudit {
  const brainDir = join(projectRoot, '.brain');
  let decayable = 0;
  let permanent = 0;
  if (existsSync(brainDir)) {
    const files = readdirSync(brainDir, { withFileTypes: true });
    for (const f of files) {
      if (!f.isFile() || !f.name.endsWith('.md')) continue;
      const lines = readFileSync(join(brainDir, f.name), 'utf-8').split('\n').length;
      if (DECAY_EXEMPT.has(f.name)) permanent += lines;
      else decayable += lines;
    }
  }
  return {
    decayableLines: decayable,
    permanentLines: permanent,
    totalLines: decayable + permanent,
    budget,
    status: decayable > budget ? 'OVER' : 'OK',
    recommendation: decayable > budget ? 'run deckent cleanup --decay' : null,
  };
}
```

Make sure `readFileSync`, `readdirSync`, `existsSync`, `join` are imported.
