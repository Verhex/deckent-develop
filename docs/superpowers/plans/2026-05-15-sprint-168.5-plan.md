```typescript
// src/orchestra/task-builder.ts içine ekle (export)
export function parseDependencyField(raw: string): string[] {
  const trimmed = raw.trim();
  if (trimmed === '') return [];

  // JSON array passthrough
  if (trimmed.startsWith('[') && trimmed.endsWith(']')) {
    try {
      const parsed = JSON.parse(trimmed);
      if (Array.isArray(parsed)) {
        return parsed.map((s: unknown) => String(s).trim()).filter(s => s.length > 0);
      }
    } catch {
      // Fallthrough to comma-split
    }
  }

  // Comma-separated or bare string
  return trimmed
    .split(',')
    .map(s => s.trim())
    .filter(s => s.length > 0);
}
```

DIRECTIVES parse fonksiyonunda mevcut `dep = raw;` (string atama) yerini `dep = parseDependencyField(raw);` ile değiştir.

```typescript
// src/orchestra/sprint-spawner.ts
export function normalizeScopeFiles(files: string[]): string[] {
  return files
    .map(f => f.replace(/^\.\//, ''))     // leading ./
    .map(f => f.replace(/\/+/g, '/'))      // double slash
    .map(f => f.replace(/\/$/, ''));       // trailing slash
}

export function detectScopeCollisions(tasks: Array<{ id: string; scope: { filesWrite: string[] } }>):
  Array<{ files: string[]; tasks: string[] }> {
  const fileToTasks = new Map<string, string[]>();
  for (const task of tasks) {
    const normalized = normalizeScopeFiles(task.scope?.filesWrite || []);
    for (const f of normalized) {
      const arr = fileToTasks.get(f) || [];
      arr.push(task.id);
      fileToTasks.set(f, arr);
    }
  }
  const collisions: Array<{ files: string[]; tasks: string[] }> = [];
  for (const [file, taskIds] of fileToTasks) {
    if (taskIds.length > 1) {
      collisions.push({ files: [file], tasks: taskIds });
    }
  }
  return collisions;
}
```

```typescript
export interface MemoryRelation {
  from_id: string;
  to_id: string;
  type: 'references' | 'supersedes' | 'caused_by' | 'resolves' | 'blocks' | 'depends_on';
  metadata?: Record<string, unknown>;
}

// src/core/memory-store.ts MemoryStore class içine:
insertRelation(rel: MemoryRelation): void {
  const fromExists = this.db.prepare('SELECT 1 FROM entries WHERE id = ?').get(rel.from_id);
  if (!fromExists) throw new Error(`Orphan reference: from_id ${rel.from_id} not found`);
  const toExists = this.db.prepare('SELECT 1 FROM entries WHERE id = ?').get(rel.to_id);
  if (!toExists) throw new Error(`Orphan reference: to_id ${rel.to_id} not found`);

  this.db.prepare(`
    INSERT OR IGNORE INTO relations (from_id, to_id, type, metadata, created_at)
    VALUES (?, ?, ?, ?, datetime('now'))
  `).run(rel.from_id, rel.to_id, rel.type, JSON.stringify(rel.metadata || {}));
}

getRelations(fromId: string): MemoryRelation[] {
  return this.db.prepare(`
    SELECT from_id, to_id, type, metadata
    FROM relations WHERE from_id = ?
  `).all(fromId) as MemoryRelation[];
}
```

```javascript
#!/usr/bin/env node
// scripts/memory/migrate-relations.mjs
import { readFileSync } from 'node:fs';
import Database from 'better-sqlite3';

const DB_PATH = '.brain/memory.db';
const ARCHIVE_PATH = '.brain/archive/pre-v2/DECISIONS.md';

const RELATION_PATTERNS = {
  references: /references?\s+(?:to\s+)?ADR[-\s]?(\d+)/gi,
  supersedes: /supersedes?\s+ADR[-\s]?(\d+)/gi,
  caused_by: /caused\s+by\s+ADR[-\s]?(\d+)/gi,
  resolves: /resolves?\s+ADR[-\s]?(\d+)/gi,
  blocks: /blocks?\s+ADR[-\s]?(\d+)/gi,
  depends_on: /depends?\s+on\s+ADR[-\s]?(\d+)/gi,
};

function main() {
  const md = readFileSync(ARCHIVE_PATH, 'utf-8');
  const db = new Database(DB_PATH);

  const sections = md.split(/^##\s+ADR[-\s]?/m).slice(1);

  let inserted = 0;
  let skipped = 0;

  for (const section of sections) {
    const idMatch = section.match(/^(\d+)/);
    if (!idMatch) continue;
    const fromId = `adr-${idMatch[1].padStart(3, '0')}`;

    const fromExists = db.prepare('SELECT 1 FROM entries WHERE id = ?').get(fromId);
    if (!fromExists) { skipped++; continue; }

    for (const [type, pattern] of Object.entries(RELATION_PATTERNS)) {
      pattern.lastIndex = 0;
      let match;
      while ((match = pattern.exec(section)) !== null) {
        const toId = `adr-${match[1].padStart(3, '0')}`;
        const toExists = db.prepare('SELECT 1 FROM entries WHERE id = ?').get(toId);
        if (!toExists) { skipped++; continue; }
        try {
          db.prepare(`
            INSERT OR IGNORE INTO relations (from_id, to_id, type, created_at)
            VALUES (?, ?, ?, datetime('now'))
          `).run(fromId, toId, type);
          inserted++;
        } catch (e) {
          console.error(`Skip ${fromId}→${toId} (${type}): ${e.message}`);
          skipped++;
        }
      }
    }
  }

  console.log(`Inserted: ${inserted}, Skipped: ${skipped}`);
  db.close();
}

main();
```

```javascript
  { name: 'AWS_SECRET', regex: /(?:aws_secret|secret_access_key)[\s'"=:]+[A-Za-z0-9/+=]{40}/gi },
  { name: 'GITHUB_PAT', regex: /gh[pousr]_[A-Za-z0-9]{36,255}/g },
  { name: 'OPENAI_KEY', regex: /sk-(?:proj-)?[A-Za-z0-9_-]{40,}/g },

  { name: 'TELEGRAM_TOKEN', regex: /\b\d{8,10}:[A-Za-z0-9_-]{35}\b/g },
  { name: 'PRIVATE_KEY', regex: /-----BEGIN (?:RSA |OPENSSH |EC |DSA )?PRIVATE KEY-----/g },
  { name: 'ENV_VALUE', regex: /^\s*[A-Z_]+_(?:TOKEN|KEY|SECRET|PASSWORD)\s*=\s*[A-Za-z0-9_-]{16,}/gm },
];

function loadBaseline() {
  if (!existsSync('.secrets-baseline')) return { allowlist: [] };
  return JSON.parse(readFileSync('.secrets-baseline', 'utf-8'));
}

function simpleHash(s) {
  let h = 0;
  for (const c of s) h = ((h << 5) - h + c.charCodeAt(0)) | 0;
  return h.toString(16);
}

function lineOf(content, idx) {
  return content.slice(0, idx).split('\n').length;
}

function scan(mode = 'check') {
  const baseline = loadBaseline();
  const trackedFiles = execFileSync('git', ['ls-files'], { encoding: 'utf-8' })
    .split('\n').filter(f => f.length > 0);

  const hits = [];
  for (const file of trackedFiles) {
    if (file.startsWith('node_modules/') || file.endsWith('.lock') || file === '.secrets-baseline') continue;
    if (file.endsWith('.png') || file.endsWith('.jpg') || file.endsWith('.svg')) continue;
    let content;
    try { content = readFileSync(file, 'utf-8'); } catch { continue; }
    for (const { name, regex } of PATTERNS) {
      regex.lastIndex = 0;
      let match;
      while ((match = regex.exec(content)) !== null) {
        const hash = simpleHash(match[0]);
        const allowed = baseline.allowlist.some(a => a.file === file && a.hash === hash);
        if (!allowed) hits.push({ file, pattern: name, line: lineOf(content, match.index), hash });
      }
    }
  }

  if (mode === 'build-baseline') {
    const allowlist = hits.map(h => ({ file: h.file, pattern: h.pattern, hash: h.hash, note: 'baseline-build' }));
    writeFileSync('.secrets-baseline', JSON.stringify({ allowlist, builtAt: new Date().toISOString() }, null, 2));
    console.log(`Baseline built: ${allowlist.length} entries.`);
    process.exit(0);
  }

  if (hits.length > 0) {
    console.error(`SECRETS DETECTED (${hits.length}):`);
    hits.forEach(h => console.error(`  ${h.file}:${h.line} [${h.pattern}]`));
    process.exit(1);
  }
  console.log('No secrets detected.');
  process.exit(0);
}
```

```typescript
  store: MemoryStore,
  exportsDir: string,
  options: { strict?: boolean } = {}

  const db = (store as any).db;
  const relationsBackup = db.prepare('SELECT * FROM relations').all();

    // ... existing parse/import logic ...

    const stmt = db.prepare(`
      INSERT OR IGNORE INTO relations (from_id, to_id, type, metadata, created_at)
      VALUES (?, ?, ?, ?, ?)
    `);
    for (const rel of relationsBackup) {
      const fromOk = db.prepare('SELECT 1 FROM entries WHERE id = ?').get(rel.from_id);
      const toOk = db.prepare('SELECT 1 FROM entries WHERE id = ?').get(rel.to_id);
      if (fromOk && toOk) {
        stmt.run(rel.from_id, rel.to_id, rel.type, rel.metadata, rel.created_at);
      }
    }

    const after = db.prepare('SELECT COUNT(*) as c FROM relations').get();
    if (options.strict && after.c < relationsBackup.length) {
      throw new Error(`Relations lost: before=${relationsBackup.length}, after=${after.c}`);

  } catch (err) {

    const stmt = db.prepare(`
      INSERT INTO relations (from_id, to_id, type, metadata, created_at)
      VALUES (?, ?, ?, ?, ?)
    `);
    for (const rel of relationsBackup) {
      try { stmt.run(rel.from_id, rel.to_id, rel.type, rel.metadata, rel.created_at); } catch {}
    }
    throw err;
  }
```

```typescript
  const date = metadata.date || (adr.created_at ? adr.created_at.slice(0, 10) : '');

  let body = `# ${id} — ${title}\n\n`;
  body += `**Status:** ${status}\n`;
  if (date) body += `**Date:** ${date}\n`;
  body += '\n';

  const content = adr.content || '';
  const present = new Set<string>();
  const sectionRegex = /##\s+(\w+)/gi;
  let m;
  while ((m = sectionRegex.exec(content)) !== null) {
    present.add(m[1]);
  }

  body += content;

  for (const section of MADR_SECTIONS) {
    if (!present.has(section)) {
      body += `\n\n## ${section}\n\n_To be backfilled in future sprint_`;
    }
  }

  body += '\n';
  return body;
}

export async function exportAdrToFs(store: any, adrDir: string = 'docs/adr'): Promise<void> {
  mkdirSync(adrDir, { recursive: true });
  const adrs = store.getByType('adr');

  for (const adr of adrs) {
    const idNum = adr.id.replace(/^adr-/i, '').padStart(3, '0');
    const slug = slugify(adr.title);
    const filename = `${idNum}-${slug}.md`;
    const filepath = join(adrDir, filename);

    if (existsSync(filepath)) {
      const fileMtime = statSync(filepath).mtimeMs;
      const dbMtime = new Date(adr.updated_at).getTime();
      if (fileMtime > dbMtime) continue;
    }

    const md = buildMadrMarkdown(adr);
    writeFileSync(filepath, md, 'utf-8');
  }
}
```
