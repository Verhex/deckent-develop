```typescript
export class DependencyCycleError extends Error {
  constructor(cycle: string[]) {
    super(`DependencyCycleError: ${cycle.join(' → ')}`);
    this.name = 'DependencyCycleError';
  }
}

export function validateDependencyGraph(tasks: Task[]): void {
  const visited = new Set<string>();
  const visiting = new Set<string>();
  const byId = new Map(tasks.map(t => [t.id, t]));

  function dfs(id: string, path: string[]): void {
    if (visited.has(id)) return;
    if (visiting.has(id)) throw new DependencyCycleError([...path, id]);
    visiting.add(id);
    const task = byId.get(id);
    for (const dep of task?.dependencies || []) {
      dfs(dep, [...path, id]);
    }
    visiting.delete(id);
    visited.add(id);
  }

  for (const t of tasks) dfs(t.id, []);
}
```

```typescript
export function stripCodeSnippets(text: string): string {
  return text.replace(/`[^`]*`/g, '').replace(/\b\w+\.\w+\([^)]*\)/g, '');
}

export function parseScopeFromLine(line: string): { directories: string[] } {
  const match = line.match(/^- Scope:\s*(.*)$/);
  if (!match) throw new Error('NotAScopeLine');
  const raw = match[1].trim();
  if (!raw) throw new Error('EmptyScope');
  const parts = raw.split(',').map(p => p.trim()).filter(Boolean);
  for (const p of parts) {
    if (p.startsWith('/')) throw new Error('AbsolutePathNotAllowed');
  }
  return { directories: parts };
}
```

```typescript
// src/orchestra/baseline-tracker.ts
import { writeFile, readFile } from 'node:fs/promises';

export interface BaselineSnapshot {
  files: number;
  pass: number;
  fail: number;
  skipped: number;
  timestamp: string;
}

export async function writeBaseline(sprintId: string, snap: BaselineSnapshot): Promise<void> {
  await writeFile(`.deckent/${sprintId}-baseline.json`, JSON.stringify(snap, null, 2));
}

export async function readBaseline(sprintId: string): Promise<BaselineSnapshot> {
  const raw = await readFile(`.deckent/${sprintId}-baseline.json`, 'utf8');
  return JSON.parse(raw);
}

export async function compareBaseline(sprintId: string, current: BaselineSnapshot): Promise<{ delta: number }> {
  const base = await readBaseline(sprintId);
  return { delta: current.fail - base.fail };
}
```

```typescript
const HONESTY_REGEX = /pre-existing\s+(failures|failing|tests)|unrelated\s+to\s+(this|my)\s+task|already\s+failing/i;

export async function detectHonestyViolation(
  result: { notes: string },
  sprintId: string,
  comparator: (sprintId: string) => Promise<{ delta: number }>
): Promise<{ flag?: 'HONESTY_VIOLATION' }> {
  if (!HONESTY_REGEX.test(result.notes)) return {};
  const { delta } = await comparator(sprintId);
  if (delta > 0) return { flag: 'HONESTY_VIOLATION' };
  return {};
}
```

```typescript
import { appendFile, readFile, mkdir } from 'node:fs/promises';
import { hrtime } from 'node:process';

const METRICS_PATH = '.deckent/metrics.jsonl';

async function ensureDir(): Promise<void> {
  try { await mkdir('.deckent', { recursive: true }); } catch {}
}

export async function metric(name: string, value: number, tags?: Record<string, string>): Promise<void> {
  await ensureDir();
  const entry = { type: 'metric', name, value, tags: tags || {}, t: Date.now() };
  await appendFile(METRICS_PATH, JSON.stringify(entry) + '\n');
}

export async function trace<T>(operation: string, fn: () => Promise<T>): Promise<T> {
  const start = hrtime.bigint();
  try {
    const result = await fn();
    const durationMs = Number((hrtime.bigint() - start) / 1000000n);
    await metric(`trace.${operation}`, durationMs);
    return result;
  } catch (err) {
    const durationMs = Number((hrtime.bigint() - start) / 1000000n);
    await ensureDir();
    const entry = { type: 'trace', operation, durationMs, error: (err as Error).message, t: Date.now() };
    await appendFile(METRICS_PATH, JSON.stringify(entry) + '\n');
    throw err;
  }
}

export async function structuredLog(
  level: 'debug' | 'info' | 'warn' | 'error',
  message: string,
  context?: Record<string, any>
): Promise<void> {
  await ensureDir();
  const pinoLevels = { debug: 20, info: 30, warn: 40, error: 50 };
  const entry = { level: pinoLevels[level], msg: message, time: Date.now(), ...context };
  await appendFile(METRICS_PATH, JSON.stringify(entry) + '\n');
}

function percentile(sorted: number[], p: number): number {
  if (sorted.length === 0) return 0;
  const idx = Math.floor((p / 100) * (sorted.length - 1));
  return sorted[idx];
}

export async function generateLoadReport(): Promise<string> {
  let raw = '';
  try { raw = await readFile(METRICS_PATH, 'utf8'); } catch { return '# Load Test Report\n\nNo metrics.\n'; }
  const entries = raw.trim().split('\n').filter(Boolean).map(l => JSON.parse(l));
  const byMetric = new Map<string, number[]>();
  for (const e of entries) {
    if (e.type === 'metric') {
      const arr = byMetric.get(e.name) || [];
      arr.push(e.value);
      byMetric.set(e.name, arr);
    }
  }
  let md = '# Load Test Report — Sprint 134\n\n';
  for (const [name, values] of byMetric.entries()) {
    const sorted = [...values].sort((a, b) => a - b);
    md += `## ${name}\n`;
    md += `- count: ${values.length}\n`;
    md += `- p50: ${percentile(sorted, 50)}\n`;
    md += `- p95: ${percentile(sorted, 95)}\n`;
    md += `- p99: ${percentile(sorted, 99)}\n\n`;
  }
  const lockWaits = byMetric.get('lock.wait_ms') || [];
  if (lockWaits.length > 0) {
    md += '## File Lock Histogram\n';
    const buckets = { '<10ms': 0, '10-50ms': 0, '50-100ms': 0, '100-500ms': 0, '>500ms': 0 };
    for (const v of lockWaits) {
      if (v < 10) buckets['<10ms']++;
      else if (v < 50) buckets['10-50ms']++;
      else if (v < 100) buckets['50-100ms']++;
      else if (v < 500) buckets['100-500ms']++;
      else buckets['>500ms']++;
    }
    for (const [k, v] of Object.entries(buckets)) md += `- ${k}: ${v}\n`;
  }
  return md;
}
```

```typescript
import { realpath } from 'node:fs/promises';
import * as path from 'node:path';

export async function checkScopeWithSymlinks(
  targetPath: string,
  scopeDirs: string[]
): Promise<boolean> {
  let resolved: string;
  try {
    resolved = await realpath(targetPath);
  } catch (err: any) {
    if (err.code === 'ELOOP') throw new Error('SymlinkCycle');
    throw err;
  }
  const absResolved = path.resolve(resolved);
  for (const dir of scopeDirs) {
    const absDir = path.resolve(dir);
    if (absResolved.startsWith(absDir)) return true;
  }
  return false;
}
```

```typescript
export function formatRubricTable(results: TaskResult[]): string {
  let md = '### Rubric Scores (Sprint 134)\n\n';
  md += '| Task | Correctness | Coverage | Scope | Docs | Avg |\n';
  md += '|------|-------------|----------|-------|------|-----|\n';
  let sprintAvgSum = 0, sprintAvgCount = 0;
  for (const r of results) {
    if (!r.rubricScores) {
      md += `| ${r.taskId} | N/A | N/A | N/A | N/A | N/A |\n`;
      continue;
    }
    const s = r.rubricScores;
    const avg = (s.correctness + s.test_coverage + s.scope_compliance + s.documentation) / 4;
    md += `| ${r.taskId} | ${s.correctness} | ${s.test_coverage} | ${s.scope_compliance} | ${s.documentation} | ${avg} |\n`;
    sprintAvgSum += avg;
    sprintAvgCount++;
  }
  if (sprintAvgCount > 0) md += `\n**Sprint Avg:** ${sprintAvgSum / sprintAvgCount}\n`;
  return md;
}
```

```typescript
export interface SelfAuditDeps {
  tscFn: () => Promise<{ status: 'PASS' | 'FAIL'; errors: string[] }>;
  vitestFn: () => Promise<{ status: 'PASS' | 'FAIL'; delta: { files: number; pass: number; fail: number; skipped: number } }>;
  honestyFn: () => Promise<{ violations: number; flaggedTasks: string[] }>;
  obsFn: () => Promise<{ metricsJsonlExists: boolean; lineCount: number }>;
}

export async function runSelfAuditGate(
  sprintId: string,
  deps: SelfAuditDeps = defaultDeps()
): Promise<SelfAuditResult> {
  const [tsc, vitest, honesty, observability] = await Promise.all([
    deps.tscFn(),
    deps.vitestFn(),
    deps.honestyFn(),
    deps.obsFn(),
  ]);

  let overallGate: 'PASS' | 'GATE_FAILURE' = 'PASS';
  if (tsc.status === 'FAIL') overallGate = 'GATE_FAILURE';
  if (vitest.status === 'FAIL') overallGate = 'GATE_FAILURE';
  if (honesty.violations > 0) overallGate = 'GATE_FAILURE';
  // observability missing → WARNING, NOT fail

  return { tsc, vitest, honesty, observability, overallGate };
}

function defaultDeps(): SelfAuditDeps {
  return {
    tscFn: async () => {
      try { execSync('npx tsc --noEmit', { timeout: 90000 }); return { status: 'PASS', errors: [] }; }
      catch (e: any) { return { status: 'FAIL', errors: [e.stdout?.toString() || e.message] }; }
    },
    vitestFn: async () => {
      try {
        const out = execSync('npx vitest run --reporter=basic', { timeout: 300000 }).toString();
        const files = parseInt(out.match(/Test Files\s+(\d+)/)?.[1] || '0');
        const pass = parseInt(out.match(/Tests\s+(\d+) passed/)?.[1] || '0');
        return { status: 'PASS', delta: { files, pass, fail: 0, skipped: 0 } };
      } catch (e: any) {
        return { status: 'FAIL', delta: { files: 0, pass: 0, fail: 1, skipped: 0 } };
      }
    },
    honestyFn: async () => ({ violations: 0, flaggedTasks: [] }),  // Task 13 may replace
    obsFn: async () => {
      try {
        const raw = await readFile('.deckent/metrics.jsonl', 'utf8');
        return { metricsJsonlExists: true, lineCount: raw.split('\n').length };
      } catch {
        return { metricsJsonlExists: false, lineCount: 0 };
      }
    },
  };
}
```
