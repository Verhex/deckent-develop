- [ ] **Step 3: Write `src/core/active-workers.ts`**

```typescript
import { readFileSync, readdirSync, existsSync } from 'node:fs';
import { join } from 'node:path';

const TASKS_DIR = '.tasks';

interface HeartbeatPayload {
  taskId?: string;
  workerId?: string;
  status?: string;
}

/**
 * Returns active worker task IDs from heartbeat files (.tasks/task-*.hb).
 * Extracted from auditor.ts:2162-2168 pattern (Sprint 168 C0e MINOR-1).
 *
 * Used by:
 * - claude.ts ProviderAdapter constructor + _cleanupOrphanedPromptFiles (Option C selective filter)
 * - auditor.ts scan loop (existing pattern)
 */
export function getActiveWorkerIds(projectRoot: string): string[] {
  const tasksDir = join(projectRoot, TASKS_DIR);
  if (!existsSync(tasksDir)) return [];

  try {
    const files = readdirSync(tasksDir);
    const ids: string[] = [];
    for (const file of files) {
      if (!file.endsWith('.hb')) continue;
      try {
        const raw = readFileSync(join(tasksDir, file), 'utf-8');
        const hb = JSON.parse(raw) as HeartbeatPayload;
        if (hb.taskId) ids.push(hb.taskId);
      } catch {
        // malformed .hb — skip
      }
    }
    return ids;
  } catch {
    return [];
  }
}
```

/**
 * Check single spawn lock state.
 * Sprint 168 C0b SpawnLock symmetric cleanup MINOR-8 (Phase 2 §141 gap).
 */
export function checkSpawnLock(projectRoot: string, filePath: string): SpawnLockInfo | null {
  const lockPath = spawnLockPathFor(projectRoot, filePath);
  if (!existsSync(lockPath)) return null;
  try {
    const raw = readFileSync(lockPath, 'utf-8');
    return JSON.parse(raw) as SpawnLockInfo;
  } catch {
    return null;
  }
}

/**
 * Check all spawn locks in projectRoot.
 */
export function checkSpawnLocks(projectRoot: string): SpawnLockInfo[] {
  const lockDir = join(projectRoot, '.locks');
  if (!existsSync(lockDir)) return [];
  const results: SpawnLockInfo[] = [];
  try {
    const files = readdirSync(lockDir);
    for (const file of files) {
      if (!file.endsWith('.spawnlock')) continue;
      try {
        const raw = readFileSync(join(lockDir, file), 'utf-8');
        results.push(JSON.parse(raw) as SpawnLockInfo);
      } catch {
        // skip malformed
      }
    }
  } catch { /* empty dir */ }
  return results;
}

/**
 * Clear stale spawn locks (TTL-based, default 5min).
 */
export function clearStaleSpawnLocks(projectRoot: string, maxAgeMs = 300000): number {
  const locks = checkSpawnLocks(projectRoot);
  const now = Date.now();
  let cleared = 0;
  for (const lock of locks) {
    const acquiredMs = new Date(lock.acquiredAt).getTime();
    if (now - acquiredMs > maxAgeMs) {
      releaseSpawnLock(projectRoot, lock.taskId, lock.filePath);
      cleared++;
      debugLog('file-lock:clearStaleSpawnLocks', `Released stale spawn lock: ${lock.filePath} (taskId=${lock.taskId})`);
    }
  }
  return cleared;
}

/**
 * Clear orphan spawn locks (taskId not in activeTaskIds).
 */
export function clearOrphanSpawnLocks(projectRoot: string, activeTaskIds: string[]): number {
  const locks = checkSpawnLocks(projectRoot);
  let cleared = 0;
  for (const lock of locks) {
    if (!activeTaskIds.includes(lock.taskId)) {
      releaseSpawnLock(projectRoot, lock.taskId, lock.filePath);
      cleared++;
      debugLog('file-lock:clearOrphanSpawnLocks', `Released orphan spawn lock: ${lock.filePath} (taskId=${lock.taskId})`);
    }
  }
  return cleared;
}

/**
 * Release stale spawn locks for specific task (on-error cleanup).
 */
export function releaseStaleSpawnLocksForTask(projectRoot: string, taskId: string): void {
  const locks = checkSpawnLocks(projectRoot);
  for (const lock of locks) {
    if (lock.taskId === taskId) {
      releaseSpawnLock(projectRoot, taskId, lock.filePath);
    }
  }
}

const BARE_TOKEN_BLOCKLIST = ['.ts', '.md', '.test', 'test.ts', '.json', '.txt'];

export interface ValidationResult {
  valid: boolean;
  errors: string[];
  sanitized: string[];
}

/**
 * Sprint 168 C0c RC1 fix — validate scope.filesWrite for bare tokens.
 */
export function validateScopeFilesWrite(filesWrite: string[]): ValidationResult {
  const errors: string[] = [];
  const sanitized: string[] = [];
  for (const fp of filesWrite) {
    if (BARE_TOKEN_BLOCKLIST.includes(fp)) {
      errors.push(`Bare token detected: ${fp}`);
      continue;
    }
    if (!fp.includes('/') && !fp.includes('\\')) {
      errors.push(`Basename without path: ${fp}`);
      continue;
    }
    sanitized.push(fp);
  }
  return { valid: errors.length === 0, errors, sanitized };
}
